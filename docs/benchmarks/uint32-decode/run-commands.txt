# Portable 1-thread run (encode, decode, batch)
export LD_LIBRARY_PATH="$NATIVE_LD"
export RAYON_NUM_THREADS=1
export TOKENIZERS_PARALLELISM=false
"$VENV_PYTHON" scripts/bench_current.py \
  --corpus "$CORPUS" \
  --model gpt2="$GPT2_JSON" \
  --model qwen2="$QWEN2_JSON" \
  --model cl100k="$CL100K_JSON" \
  --threads 1 --tasks encode,decode,batch \
  --rounds 5 --passes-encode 10 --passes-decode 10 --passes-batch 1 \
  --max-chars 1048576 --max-docs 256 \
  --batch-max-chars 16777216 --batch-max-docs 4187 \
  --tokie-wheel "$TOKIE_WHEEL" \
  --expected-tokie-wheel-sha256 "$EXPECTED_TOKIE_SHA256" \
  --wheels-dir "$WHEELS_DIR" \
  --output docs/benchmarks/uint32-decode/results-1t.json
